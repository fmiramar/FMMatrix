cmake_minimum_required(VERSION 3.15)

project(FMMatrixUGens VERSION 0.1.0 LANGUAGES CXX)

option(SCSYNTH "Build the scsynth plugin target" ON)
option(SUPERNOVA "Build the supernova plugin target" OFF)
option(STRICT "Enable strict compiler warnings" OFF)
option(FM_MATRIX_SANITIZE "Enable address and undefined-behavior sanitizers for native tests" OFF)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
    set(CMAKE_INSTALL_PREFIX "${CMAKE_BINARY_DIR}/stage" CACHE PATH "Install prefix" FORCE)
endif()

if(NOT SC_PATH)
    message(FATAL_ERROR "SC_PATH must point to a matching SuperCollider source tree")
endif()

get_filename_component(SC_PATH_FOR_MODULES "${SC_PATH}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
list(APPEND CMAKE_MODULE_PATH "${SC_PATH_FOR_MODULES}/tools/cmake_gen")
include(SuperColliderServerPlugin)
sc_do_initial_compiler_config()
sc_check_sc_path("${SC_PATH}")

set(fm_matrix_cpp_files source/FMMatrixUGens.cpp)
file(GLOB fm_matrix_sc_files CONFIGURE_DEPENDS Classes/*.sc)
file(GLOB fm_matrix_help_files CONFIGURE_DEPENDS HelpSource/Classes/*.schelp)

sc_add_server_plugin("FMMatrixUGens" "FMMatrixUGens"
    "${fm_matrix_cpp_files}" "${fm_matrix_sc_files}" "${fm_matrix_help_files}")

foreach(target FMMatrixUGens_scsynth FMMatrixUGens_supernova)
    if(TARGET ${target})
        target_include_directories(${target} SYSTEM PRIVATE
            "${SC_PATH_FOR_MODULES}/include/plugin_interface"
            "${SC_PATH_FOR_MODULES}/include/common"
            "${SC_PATH_FOR_MODULES}/common")
        target_include_directories(${target} PRIVATE source)
        if(STRICT)
            if(MSVC)
                target_compile_options(${target} PRIVATE /W4 /permissive-)
            else()
                target_compile_options(${target} PRIVATE
                    -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wno-unused-parameter)
            endif()
        endif()
    endif()
endforeach()

if(TARGET FMMatrixUGens_scsynth)
    set_target_properties(FMMatrixUGens_scsynth PROPERTIES OUTPUT_NAME "FMMatrixUGens")
endif()
if(TARGET FMMatrixUGens_supernova)
    set_target_properties(FMMatrixUGens_supernova PROPERTIES OUTPUT_NAME "FMMatrixUGens_supernova")
endif()

install(FILES README.md CHANGELOG.md LICENSE ORIGINS.md THIRD_PARTY_NOTICES.md STATUS.md
    DESTINATION FMMatrixUGens)
install(DIRECTORY notes/ DESTINATION FMMatrixUGens/notes)
install(DIRECTORY docs/ DESTINATION FMMatrixUGens/docs)
install(DIRECTORY tests/ DESTINATION FMMatrixUGens/tests)
install(DIRECTORY examples/ DESTINATION FMMatrixUGens/examples)
install(DIRECTORY benchmarks/ DESTINATION FMMatrixUGens/benchmarks
    FILES_MATCHING PATTERN "*.md" PATTERN "*.csv")
install(FILES HelpSource/Guides/FMMatrixUGens.schelp
    DESTINATION FMMatrixUGens/HelpSource/Guides)
